#include <stdio.h>   // Affichage et lecture du resultat de la commande ping.
#include <string.h>  // Recherche d'un texte avec strstr.
#include <windows.h> // Attente avec Sleep et mesure du temps avec GetTickCount.

#define IP_VM "192.168.56.101" // Adresse IPv4 de la VM, a modifier si elle change.

int main(void)
{
    printf("Ping vers %s toutes les 10 secondes. Ctrl+C pour arreter.\n", IP_VM);

    while (1) {
        DWORD debut = GetTickCount();
        char ligne[256];
        int succes = 0;

        // Un seul ping IPv4, avec une attente maximale d'une seconde.
        FILE *ping = _popen("ping.exe -4 -n 1 -w 1000 " IP_VM, "r");
        if (ping == NULL) {
            perror("Impossible de lancer le ping");
            return 1;
        }

        // Sous Windows, une reponse IPv4 reussie contient "TTL=".
        while (fgets(ligne, sizeof(ligne), ping) != NULL) {
            if (strstr(ligne, "TTL=") != NULL) {
                printf("[SUCCES] %s", ligne);
                succes = 1;
            }
        }
        _pclose(ping);

        if (!succes) {
            printf("[ECHEC] %s ne repond pas au ping.\n", IP_VM);
        }
        fflush(stdout);

        // Inclure le temps du ping dans les 10 secondes.
        DWORD duree = GetTickCount() - debut;
        if (duree < 10000) {
            Sleep(10000 - duree);
        }
    }
}
